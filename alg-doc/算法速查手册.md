| O(n²) | 100 | 10000 | 1000000 | 平方 | 冒泡、选择、插入排序 |
| O(n³) | 1000 | 1000000 | 10⁹ | 立方 | 三重循环 |
| O(2ⁿ) | 1024 | 1.27×10³⁰ | - | 指数 | 递归斐波那契、子集 |
| O(n!) | 3628800 | - | - | 阶乘 | 全排列、旅行商问题 |

### 复杂度计算规则

```javascript
// O(1) - 常数时间
function getFirst(arr) {
    return arr[0];  // 无论数组多大，都是一次操作
}

// O(n) - 线性时间
function sum(arr) {
    let total = 0;
    for (let i = 0; i < arr.length; i++) {  // n次循环
        total += arr[i];
    }
    return total;
}

// O(log n) - 对数时间
function binarySearch(arr, target) {
    let left = 0, right = arr.length - 1;
    while (left <= right) {  // 每次循环减半
        const mid = Math.floor((left + right) / 2);
        if (arr[mid] === target) return mid;
        if (arr[mid] < target) left = mid + 1;
        else right = mid - 1;
    }
    return -1;
}

// O(n log n) - 线性对数时间
function mergeSort(arr) {
    if (arr.length <= 1) return arr;
    const mid = Math.floor(arr.length / 2);
    return merge(
        mergeSort(arr.slice(0, mid)),  // log n 层递归
        mergeSort(arr.slice(mid))      // 每层合并 O(n)
    );
}

// O(n²) - 平方时间
function bubbleSort(arr) {
    for (let i = 0; i < arr.length; i++) {      // n次
        for (let j = 0; j < arr.length - i; j++) { // n次
            if (arr[j] > arr[j + 1]) {
                [arr[j], arr[j + 1]] = [arr[j + 1], arr[j]];
            }
        }
    }
    return arr;
}

// O(2ⁿ) - 指数时间
function fibonacci(n) {
    if (n <= 1) return n;
    return fibonacci(n - 1) + fibonacci(n - 2);  // 每次分裂为2个子问题
}
```

## 三、排序算法对比

### 选择建议

```
小数据集（n < 50）：插入排序
需要稳定性：归并排序
空间有限：堆排序、快速排序
已部分有序：插入排序
整数且范围小：计数排序
```

### 1. 双指针

#### 适用场景
- 有序数组、链表
- 回文检测
- 区间问题

#### 模板
```javascript
// 对撞指针
function twoPointer(arr) {
    let left = 0;
    let right = arr.length - 1;

    while (left < right) {
        // 根据条件移动指针
        if (condition) {
            left++;
        } else {
            right--;
        }
    }
}

// 快慢指针
function slowFast(head) {
    let slow = head;
    let fast = head;

    while (fast && fast.next) {
        slow = slow.next;
        fast = fast.next.next;
    }
}
```

#### 典型题目
- 两数之和（有序数组）
- 三数之和
- 环形链表检测
- 回文串验证

### 2. 滑动窗口

#### 适用场景
- 连续子数组/子串问题
- 最长/最短子串
- 包含特定字符的子串

#### 模板
```javascript
function slidingWindow(s) {
    const window = new Map();
    let left = 0;
    let right = 0;
    let result = 0;

    while (right < s.length) {
        // 扩大窗口
        const c = s[right];
        right++;
        // 更新窗口数据
        window.set(c, (window.get(c) || 0) + 1);

        // 判断是否需要缩小窗口
        while (window需要缩小) {
            // 缩小窗口
            const d = s[left];
            left++;
            // 更新窗口数据
            window.set(d, window.get(d) - 1);
        }

        // 更新结果
        result = Math.max(result, right - left);
    }

    return result;
}
```

#### 典型题目
- 无重复字符的最长子串
- 最小覆盖子串
- 找到字符串中所有字母异位词

### 3. 二分查找

#### 适用场景
- 有序数组查找
- 查找边界
- 最小化最大值、最大化最小值

#### 模板
```javascript
// 基础二分
function binarySearch(arr, target) {
    let left = 0;
    let right = arr.length - 1;

    while (left <= right) {
        const mid = left + Math.floor((right - left) / 2);

        if (arr[mid] === target) {
            return mid;
        } else if (arr[mid] < target) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }

    return -1;
}

// 查找左边界
function leftBound(arr, target) {
    let left = 0;
    let right = arr.length;

    while (left < right) {
        const mid = left + Math.floor((right - left) / 2);

        if (arr[mid] < target) {
            left = mid + 1;
        } else {
            right = mid;
        }
    }

    return left;
}

// 查找右边界
function rightBound(arr, target) {
    let left = 0;
    let right = arr.length;

    while (left < right) {
        const mid = left + Math.floor((right - left) / 2);

        if (arr[mid] <= target) {
            left = mid + 1;
        } else {
            right = mid;
        }
    }

    return left - 1;
}
```

#### 典型题目
- 搜索旋转排序数组
- 在排序数组中查找元素的第一个和最后一个位置
- 寻找峰值

### 4. DFS（深度优先搜索）

#### 适用场景
- 树/图遍历
- 连通性检测
- 路径查找

#### 模板
```javascript
// 递归版
function dfs(node, visited) {
    if (!node || visited.has(node)) return;

    // 访问节点
    visited.add(node);
    console.log(node.val);

    // 递归访问邻居
    for (const neighbor of node.neighbors) {
        dfs(neighbor, visited);
    }
}

// 迭代版（栈）
function dfsIterative(root) {
    if (!root) return;

    const stack = [root];
    const visited = new Set();

    while (stack.length > 0) {
        const node = stack.pop();

        if (visited.has(node)) continue;
        visited.add(node);

        // 访问节点
        console.log(node.val);

        // 邻居入栈（注意顺序）
        for (let i = node.neighbors.length - 1; i >= 0; i--) {
            stack.push(node.neighbors[i]);
        }
    }
}
```

#### 典型题目
- 岛屿数量
- 克隆图
- 路径总和

### 5. BFS（广度优先搜索）

#### 适用场景
- 层级遍历
- 最短路径
- 状态转换

#### 模板
```javascript
function bfs(root) {
    if (!root) return;

    const queue = [root];
    const visited = new Set([root]);

    while (queue.length > 0) {
        const levelSize = queue.length;

        // 处理当前层
        for (let i = 0; i < levelSize; i++) {
            const node = queue.shift();

            // 访问节点
            console.log(node.val);

            // 邻居入队
            for (const neighbor of node.neighbors) {
                if (!visited.has(neighbor)) {
                    visited.add(neighbor);
                    queue.push(neighbor);
                }
            }
        }
    }
}
```

#### 典型题目
- 二叉树的层序遍历
- 腐烂的橘子
- 单词接龙

### 6. 回溯

#### 适用场景
- 排列组合
- 子集问题
- 棋盘问题

#### 模板
```javascript
function backtrack(路径, 选择列表, result) {
    // 结束条件
    if (满足结束条件) {
        result.push([...路径]);
        return;
    }

    for (const 选择 of 选择列表) {
        // 剪枝
        if (不满足条件) continue;

        // 做选择
        路径.push(选择);

        // 递归
        backtrack(路径, 选择列表, result);

        // 撤销选择
        路径.pop();
    }
}
```

#### 典型题目
- 全排列
- 子集
- N皇后
- 组合总和

### 7. 动态规划

#### 识别特征
- 最优子结构
- 重叠子问题
- 求最值、计数、可行性

#### 解题步骤
1. 定义状态（dp数组含义）
2. 找状态转移方程
3. 初始化边界
4. 确定计算顺序
5. 考虑空间优化

#### 1D DP模板
```javascript
function dp1D(n) {
    const dp = new Array(n + 1);

    // 初始化
    dp[0] = 初始值;
    dp[1] = 初始值;

    // 状态转移
    for (let i = 2; i <= n; i++) {
        dp[i] = f(dp[i-1], dp[i-2], ...);
    }

    return dp[n];
}

// 空间优化
function dp1DOptimized(n) {
    let prev2 = 初始值;
    let prev1 = 初始值;

    for (let i = 2; i <= n; i++) {
        const current = f(prev1, prev2);
        prev2 = prev1;
        prev1 = current;
    }

    return prev1;
}
```

#### 2D DP模板
```javascript
function dp2D(m, n) {
    const dp = Array.from({ length: m }, () => new Array(n));

    // 初始化第一行和第一列
    for (let i = 0; i < m; i++) dp[i][0] = 初始值;
    for (let j = 0; j < n; j++) dp[0][j] = 初始值;

    // 状态转移
    for (let i = 1; i < m; i++) {
        for (let j = 1; j < n; j++) {
            dp[i][j] = f(dp[i-1][j], dp[i][j-1], ...);
        }
    }

    return dp[m-1][n-1];
}
```

#### 典型题目
- 爬楼梯
- 打家劫舍
- 最长递增子序列
- 编辑距离
- 0-1背包

### 8. 贪心

```javascript
function greedy1(arr) {
    arr.sort((a, b) => a - b);  // 根据某种标准排序

    for (const item of arr) {
        if (满足条件) {
            选择该项;
        }
    }
}

// 2. 每一步选择当前最优
function greedy2(arr) {
    let result = 0;

    for (let i = 0; i < arr.length; i++) {
        if (arr[i] > arr[i-1]) {
            result += arr[i] - arr[i-1];
        }
    }

    return result;
}
```

### 常用工具函数

```javascript
// 交换数组元素
function swap(arr, i, j) {
    [arr[i], arr[j]] = [arr[j], arr[i]];
}

// 生成范围数组
function range(start, end) {
    return Array.from({ length: end - start }, (_, i) => start + i);
}

// 填充二维数组
function create2DArray(m, n, fill = 0) {
    return Array.from({ length: m }, () => new Array(n).fill(fill));
}

// 深拷贝
function deepClone(obj, hash = new WeakMap()) {
    if (obj === null || typeof obj !== 'object') return obj;
    if (hash.has(obj)) return hash.get(obj);

    const clone = Array.isArray(obj) ? [] : {};
    hash.set(obj, clone);

    for (const key in obj) {
        if (obj.hasOwnProperty(key)) {
            clone[key] = deepClone(obj[key], hash);
        }
    }

    return clone;
}

// 防抖
function debounce(func, delay) {
    let timer = null;
    return function(...args) {
        clearTimeout(timer);
        timer = setTimeout(() => func.apply(this, args), delay);
    };
}

// 节流
function throttle(func, delay) {
    let lastTime = 0;
    return function(...args) {
        const now = Date.now();
        if (now - lastTime >= delay) {
            func.apply(this, args);
            lastTime = now;
        }
    };
}
```

## 六、面试技巧

### 解题步骤（UMPIRE）

1. **Understand（理解题意）**
   - 明确输入输出
   - 询问边界条件
   - 确认限制条件

2. **Match（匹配模式）**
   - 这是什么类型的题？
   - 用过类似的算法吗？
   - 需要什么数据结构？

3. **Plan（制定计划）**
   - 想出1-2种解法
   - 分析时间/空间复杂度
   - 选择最优方案

4. **Implement（实现代码）**
   - 边写边解释
   - 使用有意义的变量名
   - 模块化代码

5. **Review（检查代码）**
   - 走一遍测试用例
   - 检查边界条件
   - 优化代码

6. **Evaluate（分析复杂度）**
   - 时间复杂度
   - 空间复杂度
   - 可能的优化

### 常见边界条件

```javascript
// 1. 空输入
if (!arr || arr.length === 0) return 0;

// 2. 单个元素
if (arr.length === 1) return arr[0];

// 4. 越界检查

// 5. 空树/空链表
```

   - 排序

   - 提前返回
   - 跳过不必要的计算

   - 位运算

   - 规律总结

## 七、常见陷阱

### 1. 整数溢出
```javascript
// ❌ 错误：可能溢出
const mid = Math.floor((left + right) / 2);

// ✅ 正确
const mid = left + Math.floor((right - left) / 2);
```

### 2. 浮点数比较
```javascript
// ❌ 错误
if (a === b) { }

// ✅ 正确
if (Math.abs(a - b) < 1e-9) { }
```

### 3. 数组修改
```javascript
// ❌ 错误：浅拷贝
result.push(path);

// ✅ 正确：深拷贝
result.push([...path]);
```

### 4. 循环边界
```javascript
// ❌ 错误：可能越界
for (let i = 0; i <= arr.length; i++) { }

// ✅ 正确
for (let i = 0; i < arr.length; i++) { }
```

### 5. Map遍历顺序
```javascript
// Map 保持插入顺序
const map = new Map();
map.set('b', 2);
map.set('a', 1);

for (const [key, val] of map) {
    console.log(key);  // 输出：b, a
}
```
