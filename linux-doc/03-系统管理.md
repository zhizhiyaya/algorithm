## 用户和组管理

### 1. 用户管理

#### 创建用户

```bash
# 基本创建
sudo useradd username              # 创建用户
sudo useradd -m username           # 创建用户并创建主目录
sudo useradd -m -s /bin/bash username  # 指定shell

# 完整创建用户
sudo useradd -m -d /home/john -s /bin/bash -c "John Doe" -g users john

# 参数说明
# -m: 创建主目录
# -d: 指定主目录路径
# -s: 指定登录shell
# -c: 用户描述
# -g: 主组
# -G: 附加组

# 示例
$ sudo useradd -m -s /bin/bash developer
$ sudo passwd developer           # 设置密码
Enter new UNIX password: ******
Retype new UNIX password: ******
passwd: password updated successfully
```

#### 修改用户

```bash
sudo usermod -aG sudo username     # 添加用户到sudo组
sudo usermod -s /bin/zsh username  # 修改shell
sudo usermod -l newname oldname    # 重命名用户
sudo usermod -L username           # 锁定用户
sudo usermod -U username           # 解锁用户

# 示例
$ sudo usermod -aG docker,www-data developer
$ groups developer
developer : developer docker www-data

$ sudo usermod -s /bin/zsh john
$ grep john /etc/passwd
john:x:1001:1001:John Doe:/home/john:/bin/zsh
```

#### 删除用户

```bash
sudo userdel username              # 删除用户（保留主目录）
sudo userdel -r username           # 删除用户及主目录

# 示例
$ sudo userdel -r tempuser
$ ls /home/
developer  john  admin
```

```bash
id                                 # 显示当前用户信息

```

### 2. 组管理

```bash
# 创建组
sudo groupadd groupname            # 创建组
sudo groupadd -g 1500 groupname    # 指定GID创建组

# 修改组
sudo groupmod -n newname oldname   # 重命名组
sudo gpasswd -a user group         # 添加用户到组
sudo gpasswd -d user group         # 从组中删除用户

# 删除组
sudo groupdel groupname            # 删除组

# 查看组
groups                             # 显示当前用户所属组
groups username                    # 显示指定用户所属组
getent group groupname             # 显示组信息

# 示例
$ sudo groupadd developers
$ sudo gpasswd -a john developers
Adding user john to group developers

$ groups john
john : john developers

$ getent group developers
developers:x:1500:john,alice,bob
```

### 3. sudo权限管理

```bash
# 编辑sudoers文件（安全方式）
sudo visudo

# 常用配置示例
# 允许用户无密码使用sudo
username ALL=(ALL) NOPASSWD:ALL

# 只允许特定命令
username ALL=(ALL) /usr/bin/systemctl, /usr/bin/docker

# 允许组使用sudo
%groupname ALL=(ALL) ALL

# 示例
$ sudo visudo
# 添加以下行
developer ALL=(ALL) NOPASSWD:/usr/bin/systemctl, /usr/bin/docker

$ sudo -l                          # 查看当前用户sudo权限
User developer may run the following commands on this host:
    (ALL) NOPASSWD: /usr/bin/systemctl, /usr/bin/docker
```

### 2. 创建systemd服务

```bash
# 创建服务文件
sudo vim /etc/systemd/system/myapp.service
```

```ini
[Unit]
Description=My Application
After=network.target

[Service]
Type=simple
User=www-data
WorkingDirectory=/var/www/myapp
ExecStart=/usr/bin/node /var/www/myapp/app.js
Restart=always
RestartSec=10
StandardOutput=journal
StandardError=journal

[Install]
WantedBy=multi-user.target
```

```bash
# 重载systemd配置
sudo systemctl daemon-reload

# 启动并启用服务
sudo systemctl start myapp
sudo systemctl enable myapp

# 查看日志
sudo journalctl -u myapp -f
```

```bash
sudo crontab -u username -e        # 编辑指定用户crontab

# crontab格式
# 分 时 日 月 周 命令
# *  *  *  *  *  command
# │  │  │  │  │
# │  │  │  │  └─── 星期几 (0-7, 0和7都代表周日)
# │  │  │  └────── 月份 (1-12)

# │  └──────────── 小时 (0-23)
# └─────────────── 分钟 (0-59)

# 示例
$ crontab -e

# 每小时执行一次

# 每5分钟执行

# 每周一上午9点执行

# 每月1号凌晨3点执行
0 3 1 * * /home/user/scripts/monthly-cleanup.sh

# 查看cron日志
$ sudo grep CRON /var/log/syslog
Dec 23 02:00:01 server CRON[1234]: (user) CMD (/home/user/scripts/backup.sh)
```

### 3. 监控脚本示例

```bash
#!/bin/bash
# system-monitor.sh

# 检查CPU使用率
CPU_USAGE=$(top -bn1 | grep "Cpu(s)" | awk '{print $2}' | cut -d'%' -f1)
if (( $(echo "$CPU_USAGE > 80" | bc -l) )); then
    echo "WARNING: CPU usage is above 80%: $CPU_USAGE%" | mail -s "CPU Alert" admin@example.com
fi

# 检查磁盘空间
DISK_USAGE=$(df -h / | awk 'NR==2 {print $5}' | cut -d'%' -f1)
if [ $DISK_USAGE -gt 90 ]; then
    echo "WARNING: Disk usage is above 90%: $DISK_USAGE%" | mail -s "Disk Alert" admin@example.com
fi

# 检查服务状态
if ! systemctl is-active --quiet nginx; then
    echo "ERROR: Nginx is not running!" | mail -s "Service Alert" admin@example.com
    systemctl start nginx
fi
```
